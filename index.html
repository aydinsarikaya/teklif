☁️ GOOGLE DRIVE'A OTOMATİK PDF YÜKLEME AYARI

Bu özellik sayesinde hazırladığınız servis formları [ProjeAdı]_[No]_[Tarih].pdf adlandırmasıyla doğrudan Google Drive klasörünüze otomatik olarak yüklenir.

🚀 Google Drive PDF Yükleyici Kodu (Google Apps Script):
1. script.google.com adresinde "Yeni Proje" oluşturun.
2. Kod editörüne aşağıdaki kodu yapıştırıp kaydedin:
function doGet(e) {
  // Tüm cihazlar için güncel kullanıcı ve şifreleri buluttan getirir
  var usersJson = PropertiesService.getScriptProperties().getProperty('USERS_JSON');
  if (!usersJson) {
    usersJson = JSON.stringify({
      "aydin": { password: "asm", role: "admin", name: "Aydın SARIKAYA" }
    });
    PropertiesService.getScriptProperties().setProperty('USERS_JSON', usersJson);
  }
  return ContentService.createTextOutput(usersJson).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // A) KULLANICI / ŞİFRE BULUT SENKRONİZASYONU
    if (data.action === "save_users") {
      PropertiesService.getScriptProperties().setProperty('USERS_JSON', JSON.stringify(data.users));
      return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS" })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (data.action === "get_users") {
      var usersJson = PropertiesService.getScriptProperties().getProperty('USERS_JSON');
      return ContentService.createTextOutput(usersJson || "{}").setMimeType(ContentService.MimeType.JSON);
    }

    // B) PDF OLUŞTURMA, GOOGLE DRIVE'A YÜKLEME VE DOĞRUDAN MAİL GÖNDERME

    // B) PDF OLUŞTURMA VE MAİL
    var folderName = "Teknik Servis Formlari (AYDN)";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    // 2. PDF Dosyasını oluştur ve klasöre kaydet (Varsayılan olarak tamamen GİZLİ kalır)
    var bytes = Utilities.base64Decode(data.pdfBase64);
    var blob = Utilities.newBlob(bytes, 'application/pdf', data.fileName);
    var file = folder.createFile(blob);

    // 3. Müşteri Maili Girilmişse PDF'i Otomatik E-Posta Olarak Gönder
    var emailSent = false;
    var targetEmail = (data.musteriEmail || data.userEmail || "").trim();
    if (targetEmail && targetEmail.indexOf("@") !== -1) {
      try {
        var subject = "Teknik Servis ve Devreye Alma Formu - " + (data.musteriAdi || "") + " (" + (data.kayitNo || data.formNo || "") + ")";
        var bodyText = "Sayın Yetkili,\n\n" +
          "Aydın Sarıkaya Mühendislik Limited Şirketi tarafından gerçekleştirilen teknik servis ve devreye alma işlemlerine ait onaylı servis formu ekte bilginize sunulmuştur.\n\n" +
          "Proje / Müşteri: " + (data.musteriAdi || "-") + "\n" +
          "Form No: " + (data.formNo || "-") + " | Kayıt No: " + (data.kayitNo || "-") + "\n" +
          "Tarih: " + (data.tarih || "-") + "\n" +
          "Sistem: " + (data.cihazGrubu || "-") + "\n" +
          "Durum: " + (data.servisSonDurum || "-") + "\n\n" +
          "İyi çalışmalar dileriz.\n\n" +
          "AYDIN SARIKAYA MÜHENDİSLİK LTD. ŞTİ.\n" +
          "Koç Kuleleri 2A / 13 Çankaya / ANKARA\n" +
          "Tel: 0 532 654 67 14 | E-posta: info@aydn.com.tr";

        GmailApp.sendEmail(targetEmail, subject, bodyText, {
          name: "Aydın Sarıkaya Mühendislik",
          attachments: [blob]
        });
        emailSent = true;
      } catch (mailErr) {
        Logger.log("Mail gönderim hatası: " + mailErr.toString());
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "SUCCESS",
      fileUrl: file.getUrl(),
      emailSent: emailSent
    })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "ERROR",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
3. Sağ üstten Dağıt (Deploy) > Yeni Dağıtım seçin. Erişim iznini Herkes (Anyone) yapın ve Dağıt'a tıklayın.
4. Alınan Web URL'sini yukarıdaki kutucuğa yapıştırın!
